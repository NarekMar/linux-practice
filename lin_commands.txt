pwd                        # где я сейчас
ls                         # что лежит в папке
ls -la                     # то же, с подробностями и скрытыми файлами
cd имя_папки               # зайти в папку
cd ..                      # выйти на уровень выше
mkdir test                 # создать папку
touch file.txt             # создать пустой файл
echo "привет" > file.txt   # записать текст в файл
cat file.txt               # показать содержимое файла
cp file.txt copy.txt       # скопировать
mv copy.txt new.txt        # переименовать или переместить
rm new.txt                 # удалить файл

nano file.txt              # текстовый редактор в терминале (сохранить: Ctrl+O, Enter; выйти: Ctrl+X)
grep "слово" file.txt      # найти строки со словом
wc -l file.txt             # посчитать строки
head -n 2 file.txt         # первые 2 строки
tail -n 2 file.txt         # последние 2 строки
ls | wc -l                 # сколько файлов в папке (| передаёт вывод одной команды другой)
man ls                     # справка по команде (выйти: q)
ls --help                  # короткая справка
clear                      # очистить экран
